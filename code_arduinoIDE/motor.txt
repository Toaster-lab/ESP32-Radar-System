#include <ESP32Servo.h>

Servo myservo;
int pos = 0;

void setup() {
    myservo.attach(4);
}

void loop() {
    for(pos = 0; pos <= 180; pos += 2){
        myservo.write(pos);
        delay(30);
    }
    delay(500);
    for(pos = 180; pos >= 0; pos -= 2){
        myservo.write(pos);
        delay(30);
    }
    delay(500);


}
